# Tax

A tax is an event-level pricing configuration applied to products such as admission items, sessions, and other billable items.

## Example Usage

```typescript
import { Tax } from "@cvent/sdk/models/components";

let value: Tax = {
  created: new Date("2017-01-02T02:00:00Z"),
  createdBy: "hporter",
  lastModified: new Date("2019-02-12T03:00:00Z"),
  lastModifiedBy: "hporter",
  id: "00fe81ea-70e7-4b18-bff5-43167c5720c0",
  name: "Sales Tax",
  code: "TAX-001",
  calculateBy: "LOCATION",
  percentage: 8.25,
  amount: 5,
  taxSchedule: {
    id: "d6af77fe-a454-4e3c-927f-c530749a51bd",
    name: "New York State Tax Schedule",
    defaultRate: 8.875,
  },
  generalLedgerCode: {
    id: "04ca6ae2-0dc3-487b-953e-86d6abbdf7d5",
    name: "General Ledger",
    code: "General Ledger Code",
    allocation: 100,
  },
};
```
