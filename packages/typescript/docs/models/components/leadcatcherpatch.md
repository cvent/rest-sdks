# LeadCatcherPatch

Partial update payload for a lead catcher contact.

## Example Usage

```typescript
import { LeadCatcherPatch } from "@cvent/sdk/models/components";

let value: LeadCatcherPatch = {
  firstName: "Jane",
  lastName: "Smith",
  title: "Director of Sales",
  email: "jane.smith@example.com",
  phone: "+1-555-0100",
  fax: "+1-555-0101",
  address: {
    streetAddress1: "1234 Trap Road",
    streetAddress2: "Suite 100",
    city: "Vienna",
    zipPostalCode: "22182",
  },
  locale: "en-US",
  config: {
    rfpEmailEnabled: true,
    pdfLinkEnabled: true,
    supplierContactRole: "roleName",
  },
  additionalInfo: "Preferred contact for APAC venues.",
};
```
