# TravelBidSeason

Season details for a travel bid.

## Example Usage

```typescript
import { TravelBidSeason } from "@cvent/sdk/models/components";
import { RFCDate } from "@cvent/sdk/types";

let value: TravelBidSeason = {
  name: "Season 1",
  startDate: new RFCDate("2021-07-13"),
  endDate: new RFCDate("2021-09-30"),
  releasePeriod: 1,
  weekendReleasePeriod: 2,
  rates: [
    {
      propertyRoom: {
        id: "04ca6ae2-0dc3-487b-953e-86d6abbdf7d3",
      },
      singleRate: 100,
      doubleRate: 101,
      tripleRate: 102,
      quadRate: 103,
      nlraSingleRate: 104,
      nlraDoubleRate: 105,
      allotment: 100,
      status: "requested",
    },
  ],
  weekendRates: [
    {
      propertyRoom: {
        id: "04ca6ae2-0dc3-487b-953e-86d6abbdf7d3",
      },
      singleRate: 100,
      doubleRate: 101,
      tripleRate: 102,
      quadRate: 103,
      nlraSingleRate: 104,
      nlraDoubleRate: 105,
      allotment: 100,
      status: "requested",
    },
  ],
  status: "requested",
};
```
