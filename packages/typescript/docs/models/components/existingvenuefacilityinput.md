# ExistingVenueFacilityInput

An existing venue facility model with read-only generated properties.

## Example Usage

```typescript
import { ExistingVenueFacilityInput } from "@cvent/sdk/models/components";

let value: ExistingVenueFacilityInput = {
  numberOfMeetingRooms: 25,
  space: {
    totalMeetingRoomSpace: 50000,
    totalExhibitSpace: 75000,
  },
  sleeping: {
    totalRooms: 350,
    suites: 20,
    singleRooms: 200,
    doubleRooms: 130,
  },
  singleRate: {
    low: 129.99,
    high: 249.99,
  },
  doubleRate: {
    low: 149.99,
    high: 299.99,
  },
  taxRate: 8.5,
  occupancyRate: 75.5,
  yearBuilt: 1998,
  yearRenovated: 2021,
  proximityFromAirport: {
    distance: 12.5,
  },
  restrictions:
    "No outside food or beverage permitted. Music must end by 11 PM.",
  maximumCapacity: {
    seated: 500,
    standing: 800,
  },
};
```
