# TravelBidFairSeason

The fair season for a travel bid.

## Example Usage

```typescript
import { TravelBidFairSeason } from "@cvent/sdk/models/components";
import { RFCDate } from "@cvent/sdk/types";

let value: TravelBidFairSeason = {
  name: "Fair season 1",
  startDate: new RFCDate("2021-07-13"),
  endDate: new RFCDate("2021-07-13"),
  minLengthOfStay: 1,
  releasePeriod: 1,
  rates: [
    {
      propertyRoom: {
        id: "9ea95519-2fc5-49d8-991b-e954f49044fe",
      },
      singleRate: 100,
      doubleRate: 101,
      tripleRate: 102,
      quadRate: 103,
      allotment: 100,
      status: "requested",
    },
  ],
  status: "requested",
};
```
