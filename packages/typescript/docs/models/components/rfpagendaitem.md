# RfpAgendaItem

Agenda item associated with an RFP.

## Example Usage

```typescript
import { RfpAgendaItem } from "@cvent/sdk/models/components";
import { RFCDate } from "@cvent/sdk/types";

let value: RfpAgendaItem = {
  created: new Date("2017-01-02T02:00:00Z"),
  createdBy: "hporter",
  lastModified: new Date("2019-02-12T03:00:00Z"),
  lastModifiedBy: "hporter",
  name: "Agenda Item 1",
  note: "This is a sample comment for this agenda item.",
  startTime: "10:00:00",
  endTime: "10:30:00",
  roomSize: 1000,
  attendeeCount: 10000,
  roomInfoRequired: false,
  twentyFourHourHold: true,
  locatedAtPrimaryEventVenue: false,
  number: 5,
  schedules: [
    {
      dayNumber: 1,
      date: new RFCDate("2021-07-03"),
    },
  ],
  id: "f56a51a4-84b0-46c9-a612-716dc95a1c96",
  agendaBuilderAttachment: {
    id: "50dc08af-24a9-4d01-84be-3d43133dc945",
  },
};
```
