# AccountUserCustomField

Represents an account user's answer to a custom field.

## Example Usage

```typescript
import { AccountUserCustomField } from "@cvent/sdk/models/components";

let value: AccountUserCustomField = {
  id: "a1b2c3d4-e5f6-7890-abcd-ef1234567890",
  name: "What is your favorite color?",
  type: "Choice - Single Answer",
  values: [
    "Choice C",
    "Choice A",
  ],
  order: 1,
};
```
